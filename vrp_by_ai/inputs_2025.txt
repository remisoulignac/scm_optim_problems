    function fuel_consumption(slope, distance_km, load, speed_kmh){

            // calculations as per Moradi, Arts and Velazquez-Martinez - "Load Asymptotics and Dynamic Speed Optimization for the Greenest Path Problem: A Comprehensive Analysis"

            angle_radians = slope // approximation with 1% precision if slope < 0.15

            speed_ms = speed_kmh / 3.6;
            distance_m = distance_km * 1000
            //let planar = (((1/(44 * 737))* distance_m) / speed_ms)*(0.2 * 33 * 5);
            epsilon = 1; //fuel-to-air ratio
            kapa = 44; //heating value of diesel
            psi = 737; // conversion factor
            k = 0.2; // engine friction
            N = 33; // engine speed, in rps
            D = 5; // engine displacement, in liters
            P = epsilon * k * N * D / (kapa * psi);
            //planar = P * distance_m / speed_ms;

            eta = 0.45 // engine efficiency
            eta_t = 0.45 // drivetrain efficiency
            Q = epsilon/(1000 * eta * eta_t * kapa * psi);

            g = 9.81; // gravity accel
            Cr = 0.01; // coefficient of rolling resistance
            acceleration_theta = g * Math.sin(angle_radians) + g * Cr * Math.cos(angle_radians);

            ro = 1.2041; // air resistance
            S = 8; // frontal area
            Cd = 0.7; // Coefficient of aerodinamic drag
            R = epsilon * Cd * ro * S / (2000 * eta * eta_t * kapa * psi)

            //let wyav = 5500 * (1/(1000 * .45 * .45)) * acceleration_theta * speed_ms;
            //let yafv = (1/(1000* .45 * .45))* acceleration_theta * weight * speed_ms;

            //let lamda = 0.00003084
            //let kNNV = 33
            //let topography = (lamda * (kNNV + wyav + yafv + byv3) * distance_m) / speed_ms;

            w = 5000; //truck_weight, in kg
            fuel_consumed = P * distance_m / speed_ms + Math.max(0, Q * distance_m * acceleration_theta * (w + load) + R * distance_m * Math.pow(speed_ms, 2));

            return fuel_consumed;
        }

class CO2Model:
    def calculate_kg(self, distance_km, speed_kmh, load_kg, gradient_percent):
        # A realistic model would be non-linear.

        return (base_emission_per_km * load_factor * gradient_factor) * distance_km

Step 2: The Tour Cost Evaluation FunctionThis is the main objective function for the ALNS algorithm. It takes a complete tour and calculates its total CO2 cost by summing the optimal costs of its arcs.Pythondef evaluate_tour_cost(tour_permutation):
    """
    Calculates the total CO2 cost for an entire tour permutation.
    e.g., tour_permutation = [0, 5, 12, 2, ..., 30, 0]
    """
    total_co2 = 0
    current_load = total_demand  # Start with a full truck

    for i in range(len(tour_permutation) - 1):
        node_A = tour_permutation[i]
        node_B = tour_permutation[i+1]
        
        # Call our arc-cost solver from Step 1
        optimal_arc_co2 = get_optimal_arc_cost(node_A, node_B, current_load)
        
        total_co2 += optimal_arc_co2
        
        # Update the load for the *next* arc
        current_load -= demands[node_B]
        
    return total_co2

import numpy as np

# 1. Define our "State" object
class TourState(State):
    
    def __init__(self, tour):
        # tour is a list of nodes, e.g. [0, 1, 2, ..., 30, 0]
        self.tour = tour
    
    def objective(self):
        # The ALNS library calls this to get the cost
        # This is where our expensive, state-dependent
        # evaluation happens.
        return evaluate_tour_cost(self.tour)
    
    def copy(self):
        return TourState(self.tour.copy())

# 2. Define Destroy and Repair operators (example)
def random_destroy(state, random_state):
    """
    Removes 20% of customers from the tour.
    """
    tour_no_depot = state.tour[1:-1]
    n_to_remove = max(1, int(len(tour_no_depot) * 0.2))
    
    removed_nodes = []
    for _ in range(n_to_remove):
        idx = random_state.randint(0, len(tour_no_depot) - 1)
        removed_nodes.append(tour_no_depot.pop(idx))
    
    new_state = TourState([0] + tour_no_depot + [0])
    return new_state, removed_nodes

def greedy_repair(state, removed_nodes, random_state):
    """
    Re-inserts removed nodes one by one at the best (lowest CO2) position.
    """
    for node in removed_nodes:
        best_cost = float('inf')
        best_idx = -1
        
        # Try inserting 'node' into every possible position
        for idx in range(1, len(state.tour)):
            test_tour = state.tour[:idx] + [node] + state.tour[idx:]
            
            # This is slow, but correct. It re-evaluates the
            # ENTIRE tour. A faster way is "delta-evaluation".
            test_cost = evaluate_tour_cost(test_tour)
            
            if test_cost < best_cost:
                best_cost = test_cost
                best_idx = idx
        
        # Insert at the best found position
        state.tour.insert(best_idx, node)
        
    return state
